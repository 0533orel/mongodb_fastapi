# 1) בניית אימג' והעלאה (עדכן שם/טאג)
docker build -t 0533orel/data-loader-api:mongodb .
docker push 0533orel/data-loader-api:mongodb

# 2) פריסת MongoDB (כבר יש לך; אם לא - הרץ את mongo.yaml מההודעה הקודמת)
oc apply -f mongo.yaml -n 0533orel-dev
oc rollout status deploy/mongodb -n 0533orel-dev

# 3) פריסת ה-API
oc apply -f api-mongo.yaml -n 0533orel-dev
oc rollout status deploy/data-loader-api -n 0533orel-dev
